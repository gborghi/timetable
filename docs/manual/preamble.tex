%   - classe       : KOMA-Script scrbook (twoside, open=right)
%   - compilatore  : lualatex (per fontspec / OpenType)
%   - corpo        : EB Garamond (oldstyle figures, smallcaps native)
%   - sans         : Lato (per cifre tecniche, etichette, titoli di
%                    schemi tikz; usato con parsimonia)
%   - mono         : Inconsolata (codice)
%   - ornamenti    : fourier-orns + pifont (fleurons, aldine, decofour)
%   - testatine    : automark in italico, capitolo verso, sezione recto
%   - capitoli     : numero romano, fleuron, titolo in maiuscoletto,
%                    filetti orizzontali sopra e sotto
%   - capilettera  : lettrine (3 righe), prima parola in maiuscoletto
%   - bibliografia : biblatex backend biber, stile authoryear-comp
%   - indice       : imakeidx (indice analitico unico)
%
% ASCII puro nel sorgente; accenti via macro (\`e, \'e, \`a, \"u),
% virgolette ``...''.
% =====================================================================

% ---------- Capitoli (KOMA chapterlinesformat custom) ----------
% Vogliamo: filetto, parola CAPITOLO in smallcaps, numero romano
% del capitolo, fleuron centrato, titolo del capitolo in maiuscoletto
% maiuscolo, secondo filetto. Tutto centrato, niente sans.
%
% KOMA hook: \chapterlineswithprefixformat{TYPE}{PREFIX}{TITLE}
% dove PREFIX e' il blocco "Capitolo N", TITLE e' il titolo testuale.
% Lo riformuliamo del tutto.
\addtokomafont{disposition}{\rmfamily}

\setkomafont{subsection}{\rmfamily\itshape\color{cIndaco}}
\setkomafont{subsubsection}{\rmfamily\itshape\color{cInk}}
\setkomafont{paragraph}{\rmfamily\scshape\color{cInk}}

% ---------- Pull-quote ----------
\newcommand{\pullquote}[2]{%
  \par\medskip
  \noindent\hfill
  \begin{minipage}{0.78\textwidth}
    \color{cSiena}\itshape\large
    \hangindent=0.6cm\hangafter=1
    \rule{0pt}{1.4em}\textcolor{cOro}{\bfseries\Large``\,}#1\textcolor{cOro}{\bfseries\Large\,''}\\
    \upshape\small\color{cInk}\hfill --- #2
  \end{minipage}\par\medskip
}

% ---------- Figure Grok: placeholder auto-render ----------
% Uso: \grokfig{nome_file_senza_estensione}{Didascalia.}
% Se figures/<nome>.png esiste, viene incluso; altrimenti silenzio.
% Vedi docs/promptgrok.txt per la lista dei prompt e i filename.
\providecommand{\grokfig}[2]{%
  \IfFileExists{figures/#1.png}{%
    \begin{figure}[h]%
    \centering%
    \includegraphics[width=0.78\linewidth]{figures/#1.png}%
    \caption{#2}%
    \label{fig:#1}%
    \end{figure}%
  }{}%
}
% Variante stretta verticale (3:4) per termometri, clessidre, ecc.
\providecommand{\grokfigv}[2]{%
  \IfFileExists{figures/#1.png}{%
    \begin{figure}[h]%
    \centering%
    \includegraphics[height=0.45\textheight]{figures/#1.png}%
    \caption{#2}%
    \label{fig:#1}%
    \end{figure}%
  }{}%
}
